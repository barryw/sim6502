suites {
  suite("Test Suite 11 - stop_on_address feature") {
    ; Load the program and symbols
    symbols("TestPrograms/include_me_full.sym")
    load("TestPrograms/include_me_full.prg", strip_header = true)

    test("stop-on-address-numeric", "Test stop_on_address with numeric address") {
        ; Set up FillMemory parameters
        [r0L] = $00     ; Fill value
        [r1] = $1000    ; Start address
        [r2] = $10      ; Size (16 bytes)

        ; Stop inside FillMemory's loop rather than at CopyMemory. FillMemory returns
        ; before $21b8, so a stop there was never reached: the run continued past the
        ; RTS into a BRK, and fail_on_brk reported that as a failure.
        jsr($2195, stop_on_address = $21a1, fail_on_brk = true)

        ; A whole sixteen-byte fill costs about 628 cycles. Stopping partway through
        ; the first pass must cost far less, which is what proves the stop happened.
        assert(cycles < 100, "stop_on_address stopped the run early")
    }

    test("stop-on-address-symbol-target", "Test stop_on_address with symbol reference for stop address") {
        ; Set up FillMemory parameters
        [r0L] = $00     ; Fill value
        [r1] = $1000    ; Start address
        [r2] = $10      ; Size (16 bytes)

        ; A symbol in stop_on_address position, aimed at an address the routine
        ; actually reaches — its own entry point, which stops it immediately.
        jsr($2195, stop_on_address = [FillMemory], fail_on_brk = true)
        assert(cycles < 100, "stop_on_address accepted a symbol and stopped the run")
    }

    test("stop-on-address-both-symbols", "Test stop_on_address with symbols for both jsr target and stop address") {
        ; Set up FillMemory parameters
        [r0L] = $00     ; Fill value
        [r1] = $1000    ; Start address
        [r2] = $10      ; Size (16 bytes)

        ; Both the target and the stop address are symbols.
        jsr([FillMemory], stop_on_address = [FillMemory], fail_on_brk = true)
        assert(cycles < 100, "symbols work in both positions")
    }

    test("stop-on-address-nested-symbol", "Test stop_on_address with nested symbol (namespace.label)") {
        ; Set up simple state
        [r0L] = $00
        [r1] = $1000
        [r2] = $08

        ; A nested symbol parses in stop_on_address position. FillMemory never reaches
        ; Keyboard.Return, so it runs to its own RTS and on into a BRK — which is fine
        ; here, because what this test checks is that the namespaced symbol resolves.
        jsr([FillMemory], stop_on_address = [Keyboard.Return], fail_on_brk = false)
    }
  }
}
